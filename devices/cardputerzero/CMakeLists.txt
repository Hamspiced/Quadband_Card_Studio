cmake_minimum_required(VERSION 3.20)
project(quadband_cardputerzero VERSION 0.4.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(QB_DESKTOP "Build the SDL desktop preview instead of Linux framebuffer" ON)
option(QB_SANITIZERS "Enable address/undefined sanitizers for host testing" OFF)
set(QB_LVGL_DIR "" CACHE PATH "Verified LVGL source directory from dependencies.lock")
if(NOT EXISTS "${QB_LVGL_DIR}/lvgl.h")
    message(FATAL_ERROR "Run build.py to fetch the pinned LVGL source, or set QB_LVGL_DIR")
endif()
if(QB_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=address,undefined)
endif()
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
add_subdirectory("${QB_LVGL_DIR}" lvgl EXCLUDE_FROM_ALL)
find_package(Threads REQUIRED)
get_filename_component(QB_ROOT "../.." ABSOLUTE)
set(QB_PORTABLE "${QB_ROOT}/firmware/cores3/main")
file(GLOB PLATFORM_SOURCES CONFIGURE_DEPENDS src/platform/*.cpp)
add_library(zero_core STATIC src/runtime.cpp src/simulated.cpp ${PLATFORM_SOURCES}
    "${QB_PORTABLE}/bridge.cpp" "${QB_PORTABLE}/standalone/controller.cpp"
    "${QB_PORTABLE}/standalone/hf.cpp" "${QB_PORTABLE}/standalone/ndef.cpp"
    "${QB_PORTABLE}/standalone/uhf.cpp" "${QB_PORTABLE}/standalone/lf.cpp")
target_include_directories(zero_core PUBLIC src "${QB_PORTABLE}")
target_link_libraries(zero_core PUBLIC Threads::Threads)
target_compile_options(zero_core PRIVATE -Wall -Wextra -Werror)
file(GLOB UI_SOURCES CONFIGURE_DEPENDS src/ui/*.cpp)
if(UI_SOURCES)
    add_library(zero_ui STATIC ${UI_SOURCES})
    target_include_directories(zero_ui PUBLIC src "${QB_PORTABLE}")
    target_link_libraries(zero_ui PUBLIC lvgl::lvgl zero_core)
    target_compile_options(zero_ui PRIVATE -Wall -Wextra -Werror)
    add_executable(quadband-cardputerzero src/main.cpp src/display.cpp src/input.cpp)
    target_link_libraries(quadband-cardputerzero PRIVATE zero_ui zero_core)
    target_compile_definitions(quadband-cardputerzero PRIVATE QB_DESKTOP=$<BOOL:${QB_DESKTOP}>)
    target_compile_options(quadband-cardputerzero PRIVATE -Wall -Wextra -Werror)
    if(QB_DESKTOP)
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
        if(APPLE)
            pkg_get_variable(QB_SDL3_LIBDIR sdl3 libdir)
        endif()
        target_link_libraries(quadband-cardputerzero PRIVATE PkgConfig::SDL2)
    elseif(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
        message(FATAL_ERROR "The framebuffer target requires Linux; use QB_DESKTOP on this host")
    endif()
    install(TARGETS quadband-cardputerzero RUNTIME DESTINATION lib/quadband-cardputerzero)
endif()
include(CTest)
if(BUILD_TESTING)
    if(TARGET quadband-cardputerzero)
        add_test(NAME discovery_cli COMMAND quadband-cardputerzero --simulate --detect-only)
        set_tests_properties(discovery_cli PROPERTIES TIMEOUT 10 PASS_REGULAR_EXPRESSION "Reader tools connected: yes")
        if(QB_SDL3_LIBDIR)
            set_tests_properties(discovery_cli PROPERTIES ENVIRONMENT "DYLD_FALLBACK_LIBRARY_PATH=${QB_SDL3_LIBDIR}")
        endif()
    endif()
    file(GLOB ZERO_TESTS CONFIGURE_DEPENDS tests/*_test.cpp tests/platform_*.cpp tests/ui_*.cpp)
    foreach(source IN LISTS ZERO_TESTS)
        get_filename_component(name "${source}" NAME_WE)
        add_executable("${name}" "${source}")
        if(name STREQUAL "platform_pty_test")
            target_sources("${name}" PRIVATE src/platform/posix_wire.cpp src/platform/bluetooth.cpp "${QB_PORTABLE}/bridge.cpp")
            target_compile_definitions("${name}" PRIVATE QB_ZERO_PTY_TEST=1)
            target_include_directories("${name}" PRIVATE src "${QB_PORTABLE}")
            target_link_libraries("${name}" PRIVATE Threads::Threads)
        elseif(name STREQUAL "platform_linux_test")
            target_sources("${name}" PRIVATE src/platform/bluetooth.cpp "${QB_PORTABLE}/bridge.cpp")
            target_compile_definitions("${name}" PRIVATE QB_ZERO_I2C_STUB_TEST=1)
            target_include_directories("${name}" PRIVATE src "${QB_PORTABLE}")
            if(APPLE)
                target_compile_definitions("${name}" PRIVATE __linux__=1)
                target_include_directories("${name}" PRIVATE tests/platform_linux_stubs)
            endif()
            target_link_libraries("${name}" PRIVATE Threads::Threads)
        elseif(name MATCHES "^ui_" AND TARGET zero_ui)
            target_link_libraries("${name}" PRIVATE zero_ui)
        else()
            target_link_libraries("${name}" PRIVATE zero_core)
        endif()
        if(name STREQUAL "input_test")
            target_sources("${name}" PRIVATE src/input.cpp)
            target_compile_definitions("${name}" PRIVATE QB_DESKTOP=$<BOOL:${QB_DESKTOP}>)
            if(QB_DESKTOP)
                target_link_libraries("${name}" PRIVATE PkgConfig::SDL2)
            endif()
        endif()
        if(name STREQUAL "ui_integration_test")
            target_compile_definitions("${name}" PRIVATE QB_TEST_OUTPUT_DIR="${CMAKE_CURRENT_BINARY_DIR}/renders")
        endif()
        target_compile_options("${name}" PRIVATE -Wall -Wextra -Werror -UNDEBUG)
        add_test(NAME "${name}" COMMAND "${name}")
        if(name STREQUAL "bluetooth_test")
            target_compile_definitions("${name}" PRIVATE QB_BLE_FIXTURE="${CMAKE_CURRENT_SOURCE_DIR}/tests/fake_ble_helper.py")
        endif()
        set_tests_properties("${name}" PROPERTIES TIMEOUT 30)
        if(name STREQUAL "input_test" AND QB_SDL3_LIBDIR)
            set_tests_properties("${name}" PROPERTIES ENVIRONMENT "DYLD_FALLBACK_LIBRARY_PATH=${QB_SDL3_LIBDIR}")
        endif()
    endforeach()
endif()
